                p.date, 
                p.location, 
                p.organization_id,

const showProjectDetailsPage = async (req, res) => {
    try {
        const projectId = req.params.id;
        const project = await getProjectDetails(projectId);

        if (!project) {
            return res.status(404).render('errors/404', { title: 'Page Not Found' });
        }

        const title = 'Project Details';
        res.render('project', { title, project });
    } catch (error) {
        console.error(error);

    }
};

How do you see the route parameter handling in `showProjectDetailsPage` working together with your 404/500 middleware, and why did you choose to explicitly render a 404 inside the controller instead of letting the global handler do it?

User: The route parameter req.params.id captures the ID from the URL. If that ID does not exist in the database, the controller returns a 404 status right away because the route matched, but the data is missing. I chose to explicitly render the 404 inside the controller to give an immediate and controlled response for missing items. The global error handler is great for generic errors or unexpected server crashes (500), but checking the database results in the controller allows me to guide the user to the 'Page Not Found' template faster and cleaner.
